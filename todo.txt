server {
    listen 8011;
    server_name _;

    root /var/www/oltinqanot;
    index index.html;

    # Serve app
    location / {
        try_files $uri /index.html;
    }

    # Static assets caching
    location ~* \.(?:ico|css|js|gif|jpe?g|png|woff2?|eot|ttf|svg|mp4|webm)$ {
        try_files $uri /index.html;
        expires 6M;
        access_log off;
        add_header Cache-Control "public";
    }

    error_page 404 /index.html;

}
